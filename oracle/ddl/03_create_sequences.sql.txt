CREATE SEQUENCE seq_provider START WITH 1 INCREMENT BY 1 CACHE 20 NOCYCLE;
CREATE SEQUENCE seq_member START WITH 1 INCREMENT BY 1 CACHE 20 NOCYCLE;
CREATE SEQUENCE seq_fee_schedule START WITH 1 INCREMENT BY 1 CACHE 20 NOCYCLE;
CREATE SEQUENCE seq_fee_schedule_detail START WITH 1 INCREMENT BY 1 CACHE 20 NOCYCLE;
CREATE SEQUENCE seq_claim START WITH 1 INCREMENT BY 1 CACHE 100 NOCYCLE;
CREATE SEQUENCE seq_claim_detail START WITH 1 INCREMENT BY 1 CACHE 100 NOCYCLE;
CREATE SEQUENCE seq_migration_audit START WITH 1 INCREMENT BY 1 CACHE 20 NOCYCLE;

SELECT sequence_name, min_value, increment_by, cache_size, cycle_flag
FROM user_sequences
ORDER BY sequence_name;
